import tkinter as tk
from tkinter import ttk

class FemboyApp:
    def __init__(self, root):
        self.root = root
        self.root.title("Femboy App v2.0")
        self.root.geometry("400x300")
        
        label = ttk.Label(root, text="🌸 Добро пожаловать!", font=("Arial", 16))
        label.pack(pady=20)
        
        btn = ttk.Button(root, text="Нажми меня", command=self.say_hello)
        btn.pack(pady=10)
    
    def say_hello(self):
        print("Привет из FemboyApp!")

if __name__ == "__main__":
    root = tk.Tk()
    app = FemboyApp(root)
    root.mainloop()
